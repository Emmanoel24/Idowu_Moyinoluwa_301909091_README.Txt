{
  isCached: false,
  stage: 'COLLSCAN',
  filter: {
    '$and': [
      {
        office_code: {
          '$eq': 'LGW-IKO'
        }
      },
      {
        uploaded_at: {
          '$gte': ISODate('2026-10-09T22:22:07.271Z')
        }
      }
    ]
  },
  direction: 'forward'
}
